def payment_to_dict(payment):
    return {
        "payment_id": payment.payment_id,
        "booking_id": payment.booking_id,
        "invoice_id": payment.invoice_id,
        "transaction_reference": payment.transaction_reference,
        "gateway_transaction_id": payment.gateway_transaction_id,
        "amount": float(payment.amount),
        "payment_method_id": payment.payment_method_id,
        "payment_method": (
            payment.payment_method.method_name
            if payment.payment_method else None
        ),
        "payment_status_id": payment.payment_status_id,
        "payment_status": (
            payment.payment_status.status_name
            if payment.payment_status else None
        ),
        "payment_date": (
            payment.payment_date.isoformat()
            if payment.payment_date else None
        ),
        "received_by": payment.received_by,
        "notes": payment.notes,
    }


def customer_for_user(user):
    return Customer.query.filter_by(user_id=user.user_id).first()


def can_access_booking(user, booking):
    if is_staff_or_admin(user):
        return True

    customer = customer_for_user(user)

    return (
        customer is not None
        and booking.customer_id == customer.customer_id
    )


def find_payment_status(name):
    return PaymentStatus.query.filter_by(status_name=name).first()


def paid_amount_for_booking(booking_id):
    paid_status = find_payment_status("PAID")

    if not paid_status:
        raise ValueError("PAID payment status is not configured")

    total = (
        db.session.query(db.func.coalesce(db.func.sum(Payment.amount), 0))
        .filter_by(
            booking_id=booking_id,
            payment_status_id=paid_status.payment_status_id
        )
        .scalar()
    )

    return Decimal(str(total))


def outstanding_amount(booking, invoice):
    total_amount = Decimal(
        str(invoice.total_amount if invoice else booking.total_amount)
    )

    outstanding = total_amount - paid_amount_for_booking(booking.booking_id)

    return max(outstanding, Decimal("0.00"))

@payment_bp.route("/overview", methods=["GET"])
@role_required("HIGHER_ADMIN", "STAFF")
def get_payment_overview():
    bookings = Booking.query.order_by(
        Booking.booking_id.desc()
    ).all()

    paid_status = PaymentStatus.query.filter_by(
        status_name="PAID"
    ).first()

    pending_status = PaymentStatus.query.filter_by(
        status_name="PENDING"
    ).first()

    if not paid_status or not pending_status:
        return jsonify({
            "error": "PAID or PENDING payment status is not configured"
        }), 500

    results = []

    for booking in bookings:
        invoice = Invoice.query.filter_by(
        booking_id=booking.booking_id
         ).first()
        total_amount = Decimal(str(
            invoice.total_amount if invoice else booking.total_amount
        ))

        paid_total = (
            db.session.query(
                db.func.coalesce(db.func.sum(Payment.amount), 0)
            )
            .filter(
                Payment.booking_id == booking.booking_id,
                Payment.payment_status_id == paid_status.payment_status_id
            )
            .scalar()
        )
        paid_amount = Decimal(str(paid_total))
        balance = max(Decimal("0.00"), total_amount - paid_amount)

        payments = Payment.query.filter_by(
            booking_id=booking.booking_id
        ).order_by(Payment.payment_id.desc()).all()

        pending_payments = [
            payment_to_dict(payment)
            for payment in payments
            if payment.payment_status_id == pending_status.payment_status_id
        ]

        if balance <= Decimal("0.00"):
            payment_summary = "PAID"
        elif paid_amount > Decimal("0.00"):
            payment_summary = "PARTIALLY_PAID"
        elif pending_payments:
            payment_summary = "PENDING"
        else:
            payment_summary = "UNPAID"

        customer = booking.customer
        customer_name = (
            f"{customer.first_name} {customer.last_name or ''}".strip()
            if customer else "Unknown customer"
        )

        results.append({
            "booking_id": booking.booking_id,
            "booking_reference": booking.booking_reference,
            "customer_name": customer_name,
            "package_name": (
                booking.package.package_name if booking.package else None
            ),
            "travel_date": (
                booking.travel_date.isoformat() if booking.travel_date else None
            ),
            "booking_status": (
                booking.status.status_name if booking.status else None
            ),
            "invoice_id": invoice.invoice_id if invoice else None,
            "invoice_number": invoice.invoice_number if invoice else None,
            "total_amount": float(total_amount),
            "paid_amount": float(paid_amount),
            "balance": float(balance),
            "payment_summary": payment_summary,
            "pending_payments": pending_payments,
        })

    return jsonify(results), 200

# CREATE A PAYMENT REQUEST — CUSTOMER ONLY
@payment_bp.route("/", methods=["POST"])
@role_required("CUSTOMER")
def create_payment():
    user = get_current_user()

    if not user:
        return jsonify({"error": "User not found or inactive"}), 403

    data = request.get_json(silent=True) or {}

    booking_id = data.get("booking_id")
    payment_method_id = data.get("payment_method_id")

    if not booking_id or not payment_method_id:
        return jsonify({
            "error": "booking_id and payment_method_id are required"
        }), 400

    booking = Booking.query.get(booking_id)

    if not booking:
        return jsonify({"error": "Booking not found"}), 404

    if not can_access_booking(user, booking):
        return jsonify({"error": "Access denied"}), 403

    method = PaymentMethod.query.get(payment_method_id)

    if not method or method.status != "ACTIVE":
        return jsonify({"error": "Invalid or inactive payment method"}), 400

    # No real payment gateway is configured.
    if method.is_online:
        return jsonify({
            "error": "Online payments are not available yet"
        }), 400

    invoice = Invoice.query.filter_by(
        booking_id=booking.booking_id
    ).first()

    if invoice and invoice.invoice_status in ("VOID", "REFUNDED"):
        return jsonify({"error": "This invoice cannot receive payments"}), 400

    try:
        amount = outstanding_amount(booking, invoice)

        if amount <= Decimal("0.00"):
            return jsonify({"error": "No outstanding amount remains"}), 400

        pending_status = find_payment_status("PENDING")

        if not pending_status:
            return jsonify({
                "error": "PENDING payment status is not configured"
            }), 500

        payment = Payment(
            booking_id=booking.booking_id,
            invoice_id=invoice.invoice_id if invoice else None,
            transaction_reference=f"REQ-{uuid4().hex.upper()}",
            amount=amount,
            payment_method_id=method.payment_method_id,
            payment_status_id=pending_status.payment_status_id,
            notes="Payment request awaiting staff verification",
        )

        db.session.add(payment)
        db.session.commit()

        return jsonify({
            "message": "Payment request created; verification is pending",
            "payment": payment_to_dict(payment),
        }), 201

    except Exception:
        db.session.rollback()
        return jsonify({"error": "Unable to create payment request"}), 500


# UPDATE PAYMENT STATUS — STAFF/ADMIN ONLY
@payment_bp.route("/<int:payment_id>/status", methods=["PUT"])
@role_required("HIGHER_ADMIN", "STAFF")
def update_payment_status(payment_id):
    user = get_current_user()
    payment = Payment.query.get_or_404(payment_id)

    data = request.get_json(silent=True) or {}
    status_name = data.get("status_name")

    allowed_statuses = {"PAID", "FAILED", "REFUNDED"}

    if status_name not in allowed_statuses:
        return jsonify({
            "error": "status_name must be PAID, FAILED, or REFUNDED"
        }), 400

    current_status = payment.payment_status.status_name

    if current_status != "PENDING" and not (
        current_status == "PAID" and status_name == "REFUNDED"
    ):
        return jsonify({
            "error": f"Cannot change payment from {current_status} to {status_name}"
        }), 400

    new_status = find_payment_status(status_name)

    if not new_status:
        return jsonify({
            "error": f"{status_name} payment status is not configured"
        }), 500

    try:
        payment.payment_status_id = new_status.payment_status_id

        if status_name == "PAID":
            payment.received_by = user.user_id

        invoice = payment.invoice

        if invoice:
            paid_status = find_payment_status("PAID")

            paid_total = Decimal("0.00")

            if paid_status:
                paid_total = Decimal(str(
                    db.session.query(
                        db.func.coalesce(db.func.sum(Payment.amount), 0)
                    )
                    .filter(
                        Payment.invoice_id == invoice.invoice_id,
                        Payment.payment_status_id == paid_status.payment_status_id,
                    )
                    .scalar()
                ))

            invoice_total = Decimal(str(invoice.total_amount))

            if status_name == "REFUNDED":
                invoice.invoice_status = "ISSUED"
            elif paid_total >= invoice_total:
                invoice.invoice_status = "PAID"
            elif paid_total > Decimal("0.00"):
                invoice.invoice_status = "PARTIALLY_PAID"
            else:
                invoice.invoice_status = "ISSUED"

        db.session.commit()

        return jsonify({
            "message": "Payment status updated successfully",
            "payment": payment_to_dict(payment),
        })

    except Exception:
        db.session.rollback()
        return jsonify({"error": "Unable to update payment status"}), 500

def serialize_invoice(invoice, include_items=False):
    result = {
        "invoice_id": invoice.invoice_id,
        "invoice_number": invoice.invoice_number,
        "booking_id": invoice.booking_id,
        "invoice_date": (
            invoice.invoice_date.isoformat()
            if invoice.invoice_date
            else None
        ),
        "due_date": (
            invoice.due_date.isoformat()
            if invoice.due_date
            else None
        ),
        "subtotal": float(invoice.subtotal),
        "discount_amount": float(invoice.discount_amount),
        "tax_amount": float(invoice.tax_amount),
        "total_amount": float(invoice.total_amount),
        "invoice_status": invoice.invoice_status,
        "generated_by": invoice.generated_by,
    }

    if include_items:
        result["items"] = [
            {
                "invoice_item_id": item.invoice_item_id,
                "description": item.description,
                "quantity": float(item.quantity),
                "unit_price": float(item.unit_price),
                "line_total": float(item.line_total),
            }
            for item in invoice.items
        ]

    return result

# CREATE INVOICE
# HIGHER_ADMIN / STAFF ONLY
# =========================================================

@invoice_bp.route("/", methods=["POST"])
@jwt_required()
@role_required("HIGHER_ADMIN", "STAFF")
def create_invoice():
    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({
            "error": "A valid JSON object is required"
        }), 400

    booking_id = data.get("booking_id")

    if (
        isinstance(booking_id, bool)
        or not isinstance(booking_id, int)
        or booking_id <= 0
    ):
        return jsonify({
            "error": "A valid booking_id is required"
        }), 400

    booking = db.session.get(Booking, booking_id)

    if booking is None:
        return jsonify({
            "error": "Booking not found"
        }), 404

    existing_invoice = Invoice.query.filter_by(
        booking_id=booking.booking_id
    ).first()

    if existing_invoice:
        return jsonify({
            "error": "An invoice already exists for this booking",
            "invoice_id": existing_invoice.invoice_id,
            "invoice_number": existing_invoice.invoice_number,
        }), 409

    user = get_current_user()

    if not user:
        return jsonify({
            "error": "User not found or inactive"
        }), 403

     # Generate the invoice line item from the booking.
# Use the stored booking subtotal to preserve the booked price.
    items = [
       {
        "description": booking.package.package_name,
        "quantity": booking.number_of_travelers,
        "unit_price": (
            Decimal(str(booking.subtotal))
            / Decimal(str(booking.number_of_travelers))
        ),
    }
]

    try:
        # Generate an invoice number using the current year's count.
        # See the note below about concurrent invoice creation.
        current_year = datetime.now().year

        sequence_number = db.session.execute(
        text("SELECT nextval('invoice_number_seq')")
        ).scalar_one()

        invoice_number = (
         f"INV-{current_year}-{sequence_number:06d}"
)

        # Financial values come from the database booking,
        # not from client-supplied invoice totals.
        invoice = Invoice(
            invoice_number=invoice_number,
            booking_id=booking.booking_id,
            subtotal=booking.subtotal,
            discount_amount=booking.discount_amount,
            tax_amount=booking.tax_amount,
            total_amount=booking.total_amount,
            invoice_status="ISSUED",
            generated_by=user.user_id,
        )

        db.session.add(invoice)
        db.session.flush()

        items_subtotal = Decimal("0.00")
        cents = Decimal("0.01")

        for item in items:
            if not isinstance(item, dict):
                raise ValueError(
                    "Each invoice item must be an object"
                )

            description = item.get("description")

            if (
                not isinstance(description, str)
                or not description.strip()
            ):
                raise ValueError(
                    "Each invoice item needs a description"
                )

            try:
                quantity = Decimal(
                    str(item.get("quantity", "1"))
                )
                unit_price = Decimal(
                    str(item["unit_price"])
                )

                if (
                    not quantity.is_finite()
                    or not unit_price.is_finite()
                ):
                    raise ValueError(
                        "Quantity and unit price must be finite numbers"
                    )

                quantity = quantity.quantize(cents)
                unit_price = unit_price.quantize(cents)

            except (KeyError, InvalidOperation, TypeError, ValueError):
                raise ValueError(
                    "Each item needs a valid quantity and unit_price"
                )

            if not quantity.is_finite() or quantity <= 0:
                raise ValueError(
                    "Quantity must be greater than zero"
                )

            if not unit_price.is_finite() or unit_price < 0:
                raise ValueError(
                    "Unit price must be non-negative"
                )

            line_total = (quantity * unit_price).quantize(cents)
            items_subtotal += line_total

            invoice_item = InvoiceItem(
                invoice_id=invoice.invoice_id,
                description=description.strip(),
                quantity=quantity,
                unit_price=unit_price,
                line_total=line_total,
            )

            db.session.add(invoice_item)

        # If line items are supplied, their sum must equal the
        # booking subtotal before discount and tax.
        if items:
            booking_subtotal = Decimal(
                str(booking.subtotal)
            ).quantize(cents)

            if items_subtotal != booking_subtotal:
                raise ValueError(
                    "Invoice item totals must equal the booking subtotal"
                )

        db.session.commit()

        return jsonify({
            "message": "Invoice created successfully",
            "invoice_id": invoice.invoice_id,
            "invoice_number": invoice.invoice_number,
            "booking_id": invoice.booking_id,
            "total_amount": float(invoice.total_amount),
            "generated_by": invoice.generated_by,
        }), 201

    except ValueError as e:
        db.session.rollback()

        return jsonify({
            "error": str(e)
        }), 400

    except Exception:
        db.session.rollback()

        # Keep internal database and implementation details out
        # of the response. Check the Flask server log for details.
        return jsonify({
            "error": "An unexpected error occurred while creating the invoice"
        }), 500

from extensions import db


class Payment(db.Model):
    __tablename__ = "payments"

    payment_id = db.Column(
        db.BigInteger,
        primary_key=True,
        autoincrement=True
    )

    booking_id = db.Column(
        db.BigInteger,
        db.ForeignKey("bookings.booking_id"),
        nullable=False
    )

    invoice_id = db.Column(
        db.BigInteger,
        db.ForeignKey("invoices.invoice_id"),
        nullable=True
    )

    transaction_reference = db.Column(
        db.String(100),
        unique=True,
        nullable=False
    )

    gateway_transaction_id = db.Column(
        db.String(150),
        nullable=True
    )

    amount = db.Column(
        db.Numeric(12, 2),
        nullable=False
    )

    payment_method_id = db.Column(
        db.SmallInteger,
        db.ForeignKey("payment_methods.payment_method_id"),
        nullable=False
    )

    payment_status_id = db.Column(
        db.SmallInteger,
        db.ForeignKey("payment_statuses.payment_status_id"),
        nullable=False
    )

    payment_date = db.Column(
        db.DateTime,
        nullable=False,
        server_default=db.func.current_timestamp()
    )

    received_by = db.Column(
        db.BigInteger,
        db.ForeignKey("users.user_id"),
        nullable=True
    )

    notes = db.Column(
        db.Text,
        nullable=True
    )

    created_at = db.Column(
        db.DateTime,
        nullable=False,
        server_default=db.func.current_timestamp()
    )

    booking = db.relationship(
        "Booking",

                    <td className="p-4">{invoice.booking_id}</td>
                    <td className="whitespace-nowrap p-4">
                      {formatDate(invoice.invoice_date)}
                    </td>
                    <td className="whitespace-nowrap p-4">
                      {formatMoney(invoice.total_amount)}
                    </td>
                    <td className="p-4">
                      <span className="rounded-full bg-teal-50 px-3 py-1 text-xs font-medium text-teal-800">
                        {invoice.invoice_status}
                      </span>
                    </td>
                    <td className="p-4">
                      <button
                        type="button"
                        onClick={() =>
                          handleViewInvoice(invoice.invoice_id)
                        }
                        className="whitespace-nowrap font-medium text-teal-700 hover:underline"
                      >
                        View Details
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </section>

      {detailsLoading && (
        <p className="text-slate-600">Loading invoice details...</p>
      )}

      {selectedInvoice && (
        <section className="rounded-xl border border-slate-200 bg-white p-5 shadow-sm">
          <div className="flex flex-wrap items-start justify-between gap-3">
            <div>
              <h2 className="text-xl font-bold text-slate-900">
                {selectedInvoice.invoice_number}
              </h2>
              <p className="mt-1 text-sm text-slate-500">
                Booking ID: {selectedInvoice.booking_id}
              </p>
            </div>

            <button
              type="button"
              onClick={() => setSelectedInvoice(null)}
              className="rounded-lg border border-slate-300 px-3 py-2 text-sm hover:bg-slate-50"
            >
              Close Details
            </button>
          </div>

          <div className="mt-5 grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
            <div className="rounded-lg bg-slate-50 p-4">
              <p className="text-sm text-slate-500">Subtotal</p>
              <p className="mt-1 font-semibold">
                {formatMoney(selectedInvoice.subtotal)}
              </p>
            </div>

            <div className="rounded-lg bg-slate-50 p-4">
              <p className="text-sm text-slate-500">Discount</p>
              <p className="mt-1 font-semibold">
                {formatMoney(selectedInvoice.discount_amount)}
              </p>
            </div>

            <div className="rounded-lg bg-slate-50 p-4">
              <p className="text-sm text-slate-500">Tax</p>
              <p className="mt-1 font-semibold">
                {formatMoney(selectedInvoice.tax_amount)}
              </p>
            </div>

            <div className="rounded-lg bg-teal-50 p-4">
              <p className="text-sm text-teal-800">Total Amount</p>
              <p className="mt-1 font-bold text-teal-900">
                {formatMoney(selectedInvoice.total_amount)}
              </p>
            </div>
          </div>

          <div className="mt-5 grid gap-2 text-sm text-slate-600 sm:grid-cols-2">
            <p>
              Invoice date: {formatDate(selectedInvoice.invoice_date)}
            </p>
            <p>
              Due date: {formatDate(selectedInvoice.due_date)}
            </p>
            <p>Status: {selectedInvoice.invoice_status}</p>
            <p>Generated by user ID: {selectedInvoice.generated_by ?? "â€”"}</p>
          </div>

          <h3 className="mt-6 font-semibold text-slate-900">
            Invoice Items
          </h3>

          {!selectedInvoice.items?.length ? (
            <p className="mt-2 text-sm text-slate-500">
              No line items were recorded for this invoice.
            </p>
          ) : (
            <div className="mt-3 overflow-x-auto">
              <table className="w-full border-collapse text-left text-sm">
                <thead className="bg-slate-50">
                  <tr>
                    <th className="p-3">Description</th>
                    <th className="p-3">Quantity</th>
                    <th className="p-3">Unit Price</th>
                    <th className="p-3">Line Total</th>
                  </tr>
                </thead>
                <tbody>
                  {selectedInvoice.items.map((item) => (
                    <tr
                      key={item.invoice_item_id}
                      className="border-t border-slate-100"
                    >
                      <td className="p-3">{item.description}</td>
                      <td className="p-3">{item.quantity}</td>
                      <td className="p-3">{formatMoney(item.unit_price)}</td>
                      <td className="p-3">{formatMoney(item.line_total)}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </section>
      )}
    </main>
  );
}

const API_BASE_URL = "http://127.0.0.1:5000/api";

// ------------------------------
// AUTHENTICATED REQUEST HELPER
// ------------------------------

async function authenticatedRequest(path, options = {}) {
  const token = sessionStorage.getItem("access_token");

  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers: {
      ...(options.body ? { "Content-Type": "application/json" } : {}),
      ...options.headers,
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
  });

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      data.error || data.msg || `Request failed (${response.status})`
    );
  }

  return data;
}

  const token = sessionStorage.getItem("access_token");

    headers: {
      Authorization: `Bearer ${token}`,
    },

    return null;
  }

  if (!response.ok) {
    throw new Error(
      data.error || data.msg || `Request failed (${response.status})`
    );
  }

  return data.customer;
}
